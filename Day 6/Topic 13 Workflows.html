#75

Event-Driven Workflow Architecture

Workflows are built around Events and Steps. Steps are functions that consume events and emit new events. This decouples your pipeline stages and makes them independently testable.

Core concepts
Event
A Pydantic class that carries data between steps. Typed payload.
Step
An async function decorated with @step. Consumes one event, emits another.
StartEvent
The built-in trigger event. Every workflow starts here.
StopEvent
The built-in termination event. Return this to end the workflow.
Workflow
The class that holds all steps. Call await wf.run() to execute.

Minimal workflow example
from llama_index.core.workflow import (
    Workflow, StartEvent, StopEvent, Event, step
)

# 1. Define custom events (typed payloads)
class RetrievedEvent(Event):
    nodes: List[NodeWithScore]

class RankedEvent(Event):
    nodes: List[NodeWithScore]

# 2. Build the workflow
class RAGWorkflow(Workflow):

    @step
    async def retrieve(self, ev: StartEvent) -> RetrievedEvent:
        nodes = await retriever.aretrieve(ev.query)
        return RetrievedEvent(nodes=nodes)

    @step
    async def rerank(self, ev: RetrievedEvent) -> RankedEvent:
        nodes = reranker.postprocess_nodes(ev.nodes)
        return RankedEvent(nodes=nodes)

    @step
    async def synthesize(self, ev: RankedEvent) -> StopEvent:
        response = await synthesizer.asynthesize(ev.nodes)
        return StopEvent(result=str(response))

# 3. Run it
wf = RAGWorkflow(timeout=60)
result = await wf.run(query="What is our revenue forecast?")
print(result)
💡 Events are type-safe — if a step emits the wrong type, you get an error immediately. This makes workflows far easier to debug than untyped chains or pipelines.

Managing State Across Workflow Steps

Workflows use a shared Context object to pass state between steps without threading it through every event payload. Essential for accumulating results across multiple steps.

Using Context for shared state
from llama_index.core.workflow import Context

class ResearchWorkflow(Workflow):

    @step
    async def gather_sources(self, ctx: Context, ev: StartEvent) -> SourcesEvent:
        # Store data in context — available to all later steps
        await ctx.set("query", ev.query)
        await ctx.set("sources_found", 0)
        nodes = await retriever.aretrieve(ev.query)
        await ctx.set("sources_found", len(nodes))
        return SourcesEvent(nodes=nodes)

    @step
    async def analyze(self, ctx: Context, ev: SourcesEvent) -> StopEvent:
        # Read state set by a previous step
        query = await ctx.get("query")
        count = await ctx.get("sources_found")
        answer = await synthesize(query, ev.nodes)
        return StopEvent(result=f"[{count} sources] {answer}")
🗃️ Use Context for intermediate data (counts, flags, accumulated results). Use Events only for the data a step directly needs to run. Don't overload event payloads.

Branches, Loops & Conditional Logic

Workflows support dynamic routing — steps can return different event types based on logic, creating branches and loops. Use Python Union types to declare multiple possible outputs.

Conditional branching — step returns Union of events
from typing import Union

class ValidEvent(Event):   query: str
class InvalidEvent(Event): reason: str

class RouterWorkflow(Workflow):

    @step
    async def validate_query(self, ev: StartEvent) -> Union[ValidEvent, InvalidEvent]:
        if len(ev.query.strip()) < 3:
            return InvalidEvent(reason="Query too short")
        return ValidEvent(query=ev.query)

    @step
    async def process(self, ev: ValidEvent) -> StopEvent:
        result = await rag_query(ev.query)
        return StopEvent(result=result)

    @step
    async def handle_invalid(self, ev: InvalidEvent) -> StopEvent:
        return StopEvent(result=f"Invalid query: {ev.reason}")
Loops — self-correcting RAG pattern
class GradeEvent(Event):
    answer: str
    is_sufficient: bool

class CorrectiveRAGWorkflow(Workflow):

    @step
    async def grade_answer(self, ctx: Context, ev: GradeEvent) -> Union[StartEvent, StopEvent]:
        attempts = await ctx.get("attempts", default=0)

        if ev.is_sufficient or attempts >= 3:
            return StopEvent(result=ev.answer)    # ← exit loop

        # Loop back — retry with refined query
        await ctx.set("attempts", attempts + 1)
        return StartEvent(query=f"More detail on: {ev.answer}")  # ← retry
🔄 Always set a max iteration guard in loops (as shown with attempts >= 3). Workflows with no exit condition will run until they hit the timeout set on the Workflow class.

Concurrent & Parallel Step Execution

Steps that emit the same event type multiple times run in parallel automatically. Use ctx.collect_events() to gather all parallel results before continuing. This is the most powerful performance feature of Workflows.

Code
class SourceEvent(Event):
    source_type: str
    content: str

class ParallelRAGWorkflow(Workflow):

    @step
    async def search_docs(self, ev: StartEvent) -> SourceEvent:
        result = await doc_retriever.aretrieve(ev.query)
        return SourceEvent(source_type="docs", content=str(result))

    @step
    async def search_web(self, ev: StartEvent) -> SourceEvent:
        result = await web_search(ev.query)
        return SourceEvent(source_type="web", content=result)

    @step
    async def query_sql(self, ev: StartEvent) -> SourceEvent:
        result = await sql_query(ev.query)
        return SourceEvent(source_type="sql", content=result)

    @step
    async def synthesize(self, ctx: Context, ev: SourceEvent) -> StopEvent:
        # Collect all 3 SourceEvents before proceeding
        events = ctx.collect_events(ev, [SourceEvent] * 3)
        if events is None:
            return None   # wait for remaining events

        # All 3 collected — synthesize combined answer
        combined = "\n".join(e.content for e in events)
        answer = await llm.acomplete(combined)
        return StopEvent(result=str(answer))
⚡ Parallel steps run concurrently via asyncio. 3 parallel retrievals take the same time as the slowest one — not 3× longer. Huge performance win for multi-source RAG.

Human-in-the-Loop & Retry Logic

🧑 HITL in Workflows (#79)

Pause execution and wait for human input. The workflow suspends at the HITL step and resumes when the human responds — even across server restarts if using durable execution.

from llama_index.core.workflow import HumanResponseEvent

class ApprovalWorkflow(Workflow):

    @step
    async def draft_email(self, ev: StartEvent) -> HumanResponseEvent:
        draft = await write_email(ev.brief)
        # Emit HumanResponseEvent — pauses here
        return HumanResponseEvent(
            response=input(
                f"Approve this email?\n{draft}\n(yes/no): "
            )
        )

    @step
    async def send_or_cancel(self, ev: HumanResponseEvent) -> StopEvent:
        if ev.response.lower() == "yes":
            return StopEvent(result="Email sent ✓")
        return StopEvent(result="Cancelled by human")
🔁 Retry Logic (#80)

Steps can retry on failure using the @step decorator parameters or a try/except loop pattern.

import asyncio

class RobustWorkflow(Workflow):

    @step
    async def call_api(self, ev: StartEvent) -> StopEvent:
        # Manual retry with backoff
        for attempt in range(3):
            try:
                result = await flaky_api_call(ev.data)
                return StopEvent(result=result)
            except Exception as e:
                if attempt == 2:
                    raise
                # Exponential backoff
                await asyncio.sleep(2 ** attempt)
                print(f"Retry {attempt+1}: {e}")

Durable Workflows · Streaming Events · REST API Deployment

Durable workflows — pause & resume (#81)

Durable workflows persist their state between steps. If the server crashes mid-workflow, execution resumes from the last completed step — no work is lost.

from llama_index.core.workflow import Workflow

# Enable checkpointing — state saved after each step
wf = RAGWorkflow(
    timeout=300,
    checkpointer=WorkflowCheckpointer(storage="./checkpoints")
)

# If interrupted, resume from checkpoint
result = await wf.run(query="Long running analysis...")
Streaming workflow events to frontend (#82)
class ProgressEvent(Event):   message: str

class StreamingWorkflow(Workflow):

    @step
    async def retrieve(self, ctx: Context, ev: StartEvent) -> RetrievedEvent:
        # Emit progress events — visible to frontend immediately
        ctx.write_event_to_stream(ProgressEvent(message="🔍 Searching documents..."))
        nodes = await retriever.aretrieve(ev.query)
        ctx.write_event_to_stream(ProgressEvent(message=f"✅ Found {len(nodes)} sources"))
        return RetrievedEvent(nodes=nodes)

# Consume the stream
handler = wf.run(query="What are our risks?")

async for event in handler.stream_events():
    if isinstance(event, ProgressEvent):
        print(event.message)   # forward to frontend
